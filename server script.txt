(function () {
    if (input && input.action === "compareAndCreateMultiple") {
        try {
            var jdAttachmentId = saveBase64AsAttachment(input.jd);
            var jdResult = saveJobDescriptionRecord(jdAttachmentId, input.model);

            data.results = [];

            for (var i = 0; i < input.resumes.length; i++) {
                var resumeAttachmentId = saveBase64AsAttachment(input.resumes[i]);
                var gptResult = compareAttachments(resumeAttachmentId, jdAttachmentId, input.model);
                var resumeResult = saveResumeRecord(jdResult.jdSysId, resumeAttachmentId, gptResult.fitPercentage);

                data.results.push({
                    fitPercentage: gptResult.fitPercentage,
                    message: gptResult.resultText,
                    resumeSysId: resumeResult.resumeSysId,
                    resumeStatus: resumeResult.status
                });
            }

            data.success = true;

        } catch (e) {
            data.success = false;
            data.message = "Error: " + e.message;
            gs.error("compareAndCreateMultiple error: " + (e.stack || e.message));
        }
    }

    // ------------------- Save Attachment -------------------
    function saveBase64AsAttachment(fileObj) {
        var gsa = new GlideSysAttachment();
        var gr = new GlideRecord("sys_attachment");
        gr.initialize();
        var parentSysId = gr.insert(); // orphaned - attach later
        return gsa.write(
            gr,
            fileObj.name,
            "application/pdf",
            GlideStringUtil.base64DecodeAsBytes(fileObj.data)
        );
    }

    // ------------------- Save Job Description -------------------
    function saveJobDescriptionRecord(jdAttachmentSysId, modelChoice) {
        var jdText = cleanPdfText(getAttachmentText(jdAttachmentSysId));
        if (!jdText) throw new Error("Failed to extract text from Job Description PDF.");

        var extracted = extractJobDetailsWithChatGPT(jdText, modelChoice === "deepseek");

        var jdGR = new GlideRecord("u_job_description");
        jdGR.initialize();
        jdGR.u_job_title = extracted.title || "Unknown Title";
        jdGR.u_required_skills = extracted.skills || "";
        jdGR.u_experience = extracted.experience || "";
        jdGR.u_education_level = extracted.education || "";
        var jdSysId = jdGR.insert();

        return { jdSysId: jdSysId, details: extracted };
    }

    function extractJobDetailsWithChatGPT(jdText, deepseekMode) {
			//gs.addInfoMessage("jdText:"+jdText)
        var apiKey = gs.getProperty("openai.api.key");
        if (!apiKey) throw new Error("Missing system property 'openai.api.key'.");

        var r = new sn_ws.RESTMessageV2();
        r.setHttpMethod("post");
        r.setEndpoint("https://api.openai.com/v1/chat/completions");
        r.setRequestHeader("Authorization", "Bearer " + apiKey);
        r.setRequestHeader("Content-Type", "application/json");

        var instruction =
            "Extract the following fields from the job description:\n" +
            "1. Job Title\n2. Required Skills (comma-separated)\n" +
            "3. Expected Years of Experience\n4. Minimum Education Level\n\n" +
            "Return ONLY valid JSON:\n" +
            "{ \"title\": \"...\", \"skills\": \"...\", \"experience\": \"...\", \"education\": \"...\" }";

        if (deepseekMode) {
            instruction += "\nRespond in DeepSeek reasoning style, but final output MUST be valid JSON.";
        }

        var body = {
            model: "gpt-4.1",
            messages: [
                { role: "user", content: instruction },
                { role: "user", content: jdText }
            ],
            temperature: 0.2
        };

        r.setRequestBody(JSON.stringify(body));
        var res = r.execute();
        var httpStatus = res.getStatusCode();
        var responseBody = res.getBody();
		//	gs.addInfoMessage("responseBody:"+responseBody);
        if (httpStatus !== 200) {
            gs.error("Job details extraction failed (" + httpStatus + "): " + responseBody);
            return {};
        }

        try {
            var parsed = new global.JSON().decode(responseBody);
            var jsonOutput = parsed.choices[0].message.content;
            return new global.JSON().decode(jsonOutput);
        } catch (e) {
            gs.error("Failed to parse job extraction response: " + e.message + " body=" + responseBody);
            return {};
        }
    }

    // ------------------- Save Resume -------------------
    function saveResumeRecord(jobDescriptionSysId, resumeAttachmentSysId, fitPercentage) {
        var status;
        if (fitPercentage >= 70) status = "Eligible";
        else if (fitPercentage >= 50) status = "Need Review";
        else status = "Not Eligible";

        var resumeGR = new GlideRecord("u_resumes");
        resumeGR.initialize();
        resumeGR.u_job_description = jobDescriptionSysId;
        resumeGR.u_resume_attachment = resumeAttachmentSysId;
        resumeGR.u_fit_percentage = fitPercentage;
        resumeGR.u_status = status;
        var resumeSysId = resumeGR.insert();

        return { resumeSysId: resumeSysId, status: status };
    }

    // ------------------- Compare Attachments -------------------
    function compareAttachments(resumeSysId, jdSysId, modelChoice) {
        var resumeText = cleanPdfText(getAttachmentText(resumeSysId));
        var jobText = cleanPdfText(getAttachmentText(jdSysId));

        if (!resumeText) throw new Error("Failed to extract text from Resume PDF.");
        if (!jobText) throw new Error("Failed to extract text from Job Description PDF.");

        var result = compareTextsWithChatGPT(resumeText, jobText, modelChoice === "deepseek");
        var fitPercentage = extractPercentage(result);

        return {
            resultText: result,
            fitPercentage: fitPercentage
        };
    }

    // ------------------- PDF Extraction -------------------
    function getAttachmentText(sysId) {
        var gr = new GlideRecord("sys_attachment");
        if (!gr.get(sysId)) {
          //  gs.addErrorMessage("Attachment not found: " + sysId);
            return "";
        }
        return extractPdfTextViaGemini(gr);
    }

    function extractPdfTextViaGemini(sysAttachmentGR) {
        var rawBytes = new GlideSysAttachment().getBytes(sysAttachmentGR);
        if (!rawBytes) return "";

        var apiKey = gs.getProperty("google.gemini.api.key");
        if (!apiKey) throw new Error("Missing system property 'google.gemini.api.key'.");
		//	gs.addInfoMessage("apiKey:"+apiKey);

        var model = gs.getProperty("google.gemini.model", "gemini-3-flash-preview");
        var endpoint = "https://generativelanguage.googleapis.com/v1beta/models/" + model + ":generateContent?key=" + apiKey;

        var b64 = new Packages.java.lang.String(
            Packages.org.apache.commons.codec.binary.Base64.encodeBase64(rawBytes)
        ).toString();

        var payload = {
            contents: [
                {
                    parts: [
                        {
                            inlineData: { mimeType: "application/pdf", data: b64 }
                        },
                        { text: "Extract all text from this PDF and return as plain text." }
                    ]
                }
            ]
        };

        var req = new sn_ws.RESTMessageV2();
        req.setHttpMethod("POST");
        req.setEndpoint(endpoint);
        req.setRequestHeader("Content-Type", "application/json");
        req.setRequestBody(JSON.stringify(payload));

        var res = req.execute();
        var status = res.getStatusCode();
        var body = res.getBody();
//gs.addInfoMessage("status:"+status);
        if (status !== 200) {
            gs.addErrorMessage("Gemini extract failed (" + status + "): " + body);
            return "";
        }

        try {
            var parsed = new global.JSON().decode(body);
            return parsed.candidates[0].content.parts[0].text || "";
        } catch (e) {
            gs.error("Failed to parse Gemini response: " + e.message + " body=" + body);
            return "";
        }
    }

    function cleanPdfText(rawText) {
        if (!rawText) return "";
        return rawText.replace(/\r?\n+/g, " ").replace(/\s{2,}/g, " ").trim();
    }

    // ------------------- GPT Compare -------------------
    function compareTextsWithChatGPT(resume, job, deepseekMode) {
        var apiKey = gs.getProperty("openai.api.key");
        if (!apiKey) throw new Error("Missing system property 'openai.api.key'.");

        var r = new sn_ws.RESTMessageV2();
        r.setHttpMethod("post");
        r.setEndpoint("https://api.openai.com/v1/chat/completions");
        r.setRequestHeader("Authorization", "Bearer " + apiKey);
        r.setRequestHeader("Content-Type", "application/json");

        var baseInstruction =
            "I extracted text from a resume and job description. Please:\n" +
            "1. Reformat the text for readability.\n" +
            "2. Compare the resume against the job description.\n" +
            "3. Explain how well the candidate fits, noting strengths and gaps.\n" +
            "4. Provide a fit percentage (0-100) as:\nFIT PERCENTAGE: <number>";

        if (deepseekMode) {
            baseInstruction += "\n\n⚠️ Respond in DeepSeek reasoning style.";
        }

        var body = {
            model: "gpt-4.1",
            messages: [
                {
                    role: "user",
                    content: [
                        { type: "text", text: baseInstruction },
                        { type: "text", text: "Resume:\n" + resume },
                        { type: "text", text: "Job Description:\n" + job }
                    ]
                }
            ]
        };

        r.setRequestBody(JSON.stringify(body));
        var response = r.execute();
        var httpStatus = response.getStatusCode();
        var responseBody = response.getBody();

        if (httpStatus == 200) {
            var json = new global.JSON().decode(responseBody);
            return json.choices && json.choices[0] && json.choices[0].message
                ? json.choices[0].message.content
                : "";
        }

        gs.error("OpenAI compare failed (" + httpStatus + "): " + responseBody);
        return "";
    }

    // ------------------- Utils -------------------
    function extractPercentage(responseText) {
        if (!responseText) return 0;
        var match = responseText.match(/FIT PERCENTAGE:\s*([0-9]{1,3}(?:\.[0-9]+)?)/i);
        return match && match[1]
            ? Math.max(0, Math.min(100, Math.round(parseFloat(match[1]))))
            : 0;
    }
})();
